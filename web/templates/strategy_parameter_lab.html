function applyGaCustomBounds(paramRanges) { const map = { dcaRe: 'dca_rearm_drawdown_pct', stageRe: 'sell_stage_rearm_drawdown_pct', step: 'step_pct', alloc: 'equal_slice_allocation_pct', gridStep: 'grid_rebound_step_pct', gridSell: 'grid_sell_pct', minProfit: 'sell_min_profit_pct', repairCooldown: 'repair_sell_cooldown_days' }; const bounds = paramRanges.bounds || {}; for (const [suffix, field] of Object.entries(map)) { const lo = document.getElementById('gaBound_' + suffix); const hi = document.getElementById('gaBound_' + suffix + 'Max'); if (!lo || !hi) continue; const loVal = parseFloat(lo.value); const hiVal = parseFloat(hi.value); if (!isNaN(loVal) && lo.value !== '' && !isNaN(hiVal) && hi.value !== '') { bounds[field] = [loVal, hiVal]; } } }
